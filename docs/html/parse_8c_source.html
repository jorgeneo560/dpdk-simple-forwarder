32static enum parse_result parse_ipv4(const uint8_t *ip, size_t avail,
33 struct pkt_info *info) {
34 size_t ihl, total;
35
36 if (avail < IPV4_MIN_HDR || (ip[0] >> 4) != 4) {
37 return PARSE_MALFORMED;
38 }
39 ihl = (size_t)(ip[0] & 0x0f) * 4;
40 total = read_be16(ip + 2);
41 // total may be below avail: Ethernet pads short frames.
42 if (ihl < IPV4_MIN_HDR || ihl > avail || total < ihl || total > avail) {
43 return PARSE_MALFORMED;
44 }
45
46 info->family = 4;
47 info->proto = ip[9];
48 memcpy(info->src_ip.v4, ip + 12, 4);
49 memcpy(info->dst_ip.v4, ip + 16, 4);
50
51 if ((read_be16(ip + 6) & 0x1fff) != 0) {
52 return PARSE_IP;
53 }
54 if (info->proto == IPPROTO_TCP || info->proto == IPPROTO_UDP) {
55 return read_ports(ip + ihl, total - ihl, info);
56 }
57 return PARSE_IP;
58}

60static enum parse_result parse_ipv6(const uint8_t *ip, size_t avail,
61 struct pkt_info *info) {
62 size_t payload, l4_avail;
63
64 if (avail < IPV6_HDR_LEN || (ip[0] >> 4) != 6) {
65 return PARSE_MALFORMED;
66 }
67 payload = read_be16(ip + 4);
68 if (IPV6_HDR_LEN + payload > avail) {
69 return PARSE_MALFORMED;
70 }
71
72 info->family = 6;
73 info->proto = ip[6];
74 memcpy(info->src_ip.v6, ip + 8, 16);
75 memcpy(info->dst_ip.v6, ip + 24, 16);
76
77 if (info->proto == IPPROTO_TCP || info->proto == IPPROTO_UDP) {
78 // A payload length of 0 means a jumbogram, which is not parsed further than
79 // the ports.
80 l4_avail = payload != 0 ? payload : avail - IPV6_HDR_LEN;
81 return read_ports(ip + IPV6_HDR_LEN, l4_avail, info);
82 }
83 return PARSE_IP;
84}

86enum parse_result parse_packet(const uint8_t *data, size_t len,
87 struct pkt_info *info) {
88 size_t off = ETH_HDR_LEN;
89 uint16_t type;
90 int tags = 0;
91
92 memset(info, 0, sizeof(*info));
93 if (len < ETH_HDR_LEN) {
94 return PARSE_MALFORMED;
95 }
96
97 type = read_be16(data + 12);
98 while (type == ETHERTYPE_VLAN || type == ETHERTYPE_QINQ) {
99 if (tags++ == MAX_VLAN_TAGS || len < off + VLAN_TAG_LEN) {
100 return PARSE_MALFORMED;
101 }
102 type = read_be16(data + off + 2);
103 off += VLAN_TAG_LEN;
104 }
105 info->ether_type = type;
106
107 switch (type) {
108 case ETHERTYPE_IPV4:
109 return parse_ipv4(data + off, len - off, info);
110 case ETHERTYPE_IPV6:
111 return parse_ipv6(data + off, len - off, info);
112 case ETHERTYPE_ARP:
113 return PARSE_ARP;
114 default:
115 return PARSE_OTHER;
116 }

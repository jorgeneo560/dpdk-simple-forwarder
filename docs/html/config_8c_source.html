103static char *trim(char *s) {
104 char *end;
105
106 while (isspace((unsigned char)*s)) {
107 s++;
108 }
109 end = s + strlen(s);
110 while (end > s && isspace((unsigned char)end[-1])) {
111 end--;
112 }
113 *end = '\0';
114 return s;
115}

149static bool parse_mac(const char *text, uint8_t *mac) {
150 if (strlen(text) != 17) {
151 return false;
152 }
153 for (int i = 0; i < 6; i++) {
154 const char *p = text + i * 3;
155 char byte[3] = {p[0], p[1], '\0'};
156
157 if (!isxdigit((unsigned char)p[0]) || !isxdigit((unsigned char)p[1]) ||
158 (i < 5 && p[2] != ':')) {
159 return false;
160 }
161 mac[i] = (uint8_t)strtoul(byte, NULL, 16);
162 }
163 return true;
164}

177static bool apply_option(const char *path, int line, const char *section,
178 const char *key, const char *val,
179 struct app_config *cfg) {
180 for (size_t i = 0; i < NB_OPTIONS; i++) {
181 const struct option *o = &options[i];
182 char *end;
183 unsigned long v;
184
185 if (strcmp(section, o->section) != 0 || strcmp(key, o->key) != 0) {
186 continue;
187 }
188
189 errno = 0;
190 v = strtoul(val, &end, o->base);
191 if (!isxdigit((unsigned char)*val) || *end != '\0' || errno != 0 ||
192 v < o->min || v > o->max) {
193 fprintf(stderr,
194 o->base == 16
195 ? "%s:%d: invalid value '%s' for %s (expected 0x%lx-0x%lx)\n"
196 : "%s:%d: invalid value '%s' for %s (expected %lu-%lu)\n",
197 path, line, val, key, o->min, o->max);
198 return false;
199 }
200 store_value(o, cfg, v);
201 return true;
202 }
203
204 for (size_t i = 0; i < NB_MAC_OPTIONS; i++) {
205 const struct mac_option *o = &mac_options[i];
206
207 if (strcmp(section, o->section) != 0 || strcmp(key, o->key) != 0) {
208 continue;
209 }
210 if (!parse_mac(val, (uint8_t *)cfg + o->offset)) {
211 fprintf(stderr,
212 "%s:%d: invalid value '%s' for %s (expected aa:bb:cc:dd:ee:ff)\n",
213 path, line, val, key);
214 return false;
215 }
216 return true;
217 }
218
219 for (size_t i = 0; i < NB_STRING_OPTIONS; i++) {
220 const struct string_option *o = &string_options[i];
221
222 if (strcmp(section, o->section) != 0 || strcmp(key, o->key) != 0) {
223 continue;
224 }
225 if (*val == '\0' || strlen(val) >= o->size) {
226 fprintf(stderr,
227 "%s:%d: invalid value for %s (expected 1-%zu characters)\n", path,
228 line, key, o->size - 1);
229 return false;
230 }
231 snprintf((char *)cfg + o->offset, o->size, "%s", val);
232 return true;
233 }
234
235 fprintf(stderr, "%s:%d: unknown option '%s' in section [%s]\n", path, line,
236 key, section);
237 return false;
238}

240static bool load_file(const char *path, struct app_config *cfg) {
241 char buf[256], section[MAX_SECTION_LEN] = "";
242 int line = 0;
243 bool ok = true;
244 FILE *f = fopen(path, "r");
245
246 if (f == NULL) {
247 fprintf(stderr,
248 "cannot open config file %s: %s (setup.sh copies it next to the "
249 "binary)\n",
250 path, strerror(errno));
251 return false;
252 }
253
254 set_defaults(cfg);
255
256 while (fgets(buf, sizeof(buf), f) != NULL) {
257 char *hash, *eq, *key, *val;
258
259 line++;
260 if (strchr(buf, '\n') == NULL && !feof(f)) {
261 fprintf(stderr, "%s:%d: line too long\n", path, line);
262 ok = false;
263 break;
264 }
265
266 hash = strchr(buf, '#');
267 if (hash != NULL) {
268 *hash = '\0';
269 }
270 key = trim(buf);
271 if (*key == '\0') {
272 continue;
273 }
274
275 if (*key == '[') {
276 size_t n = strlen(key);
277
278 if (n < 3 || key[n - 1] != ']' || n - 2 >= sizeof(section)) {
279 fprintf(stderr, "%s:%d: malformed section header\n", path, line);
280 ok = false;
281 break;
282 }
283 key[n - 1] = '\0';
284 snprintf(section, sizeof(section), "%s", trim(key + 1));
285 if (!is_known_section(section)) {
286 fprintf(stderr, "%s:%d: unknown section [%s]\n", path, line, section);
287 ok = false;
288 break;
289 }
290 continue;
291 }
292
293 eq = strchr(key, '=');
294 if (eq == NULL) {
295 fprintf(stderr, "%s:%d: expected 'key = value' or '[section]'\n", path,
296 line);
297 ok = false;
298 break;
299 }
300 if (section[0] == '\0') {
301 fprintf(stderr, "%s:%d: option outside of a [section]\n", path, line);
302 ok = false;
303 break;
304 }
305 *eq = '\0';
306 val = trim(eq + 1);
307 key = trim(key);
308
309 if (!apply_option(path, line, section, key, val, cfg)) {
310 ok = false;
311 break;
312 }
313 }
314
315 fclose(f);
316 // The cleaner removes a flow after the timeout. A longer timeout than the
317 // export interval guarantees an export of its final counters first.
318 if (ok && cfg->flow_timeout <= cfg->export_interval) {
319 fprintf(stderr,
320 "%s: [flows] timeout (%us) must be larger than [export] interval "
321 "(%us), so a flow is exported after its last packet\n",
322 path, cfg->flow_timeout, cfg->export_interval);
323 ok = false;
324 }
325 return ok;
326}

328static bool default_path(char *buf, size_t len) {
329 char binary[PATH_MAX];
330 ssize_t n =
331 readlink("/proc/self/exe", binary,
332 sizeof(binary) - 1); // get the path of the current executable
333
334 if (n < 0 || (size_t)n == sizeof(binary) - 1) {
335 return false;
336 }
337
338 char *slash;
339 binary[n] = '\0';
340 slash = strrchr(binary, '/');
341 if (slash == NULL) {
342 return false;
343 }
344
345 *slash = '\0';
346 return (size_t)snprintf(buf, len, "%s/%s", binary, CONFIG_FILE_NAME) < len;
347}

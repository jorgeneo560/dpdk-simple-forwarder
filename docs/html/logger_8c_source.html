44// Called by stdio from whichever thread logs: only copies the text into the
45// queue.
46static ssize_t stream_write(void *cookie, const char *buf, size_t size) {
47 struct log_msg *msg;
48
49 (void)cookie;
50 pthread_mutex_lock(&lock);
51 if (!accepting) {
52 pthread_mutex_unlock(&lock);
53 fwrite(buf, 1, size, stderr);
54 return (ssize_t)size;
55 }
56 if (q_count == QUEUE_LEN) {
57 dropped++;
58 pthread_mutex_unlock(&lock);
59 return (ssize_t)size;
60 }
61
62 msg = &queue[q_tail];
63 clock_gettime(CLOCK_REALTIME, &msg->ts);
64 msg->lcore = rte_lcore_id();
65 msg->len = size < TEXT_LEN ? size : TEXT_LEN - 1;
66 memcpy(msg->text, buf, msg->len);
67 if (size >= TEXT_LEN) {
68 msg->text[msg->len - 1] = '\n'; // a line cut to fit the slot still ends
69 }
70 q_tail = (q_tail + 1) % QUEUE_LEN;
71 q_count++;
72 pthread_cond_signal(&wake);
73 pthread_mutex_unlock(&lock);
74 return (ssize_t)size;
75}

78 char timestamp[32], source[24];

81 localtime_r(&msg->ts.tv_sec, &local_time);

98static void *logger_main(void *arg) {
99 struct log_msg msg;
100
101 (void)arg;
102 for (;;) {
103 pthread_mutex_lock(&lock);
104 while (q_count == 0 && !stopping) {
105 pthread_cond_wait(&wake, &lock);
106 }
107 if (q_count == 0) {
108 accepting = false;
109 pthread_mutex_unlock(&lock);
110 return NULL;
111 }
112 msg = queue[q_head];
113 q_head = (q_head + 1) % QUEUE_LEN;
114 q_count--;
115 pthread_mutex_unlock(&lock);
116
117 write_msg(&msg);
118 }
119}

156 if (err != 0) {

158 fclose(stream);
159 fclose(log_file);

161 return false;
162 }
163 pthread_setname_np(thread, "flowfwd-log");
164 started = true;
165
166 // Only our own log type follows the configured level; DPDK libraries keep
167 // theirs.

169 rte_log_set_level(RTE_LOGTYPE_FLOWFWD, levels[level]);
170 rte_openlog_stream(stream);
171 atexit(logger_stop);
172
173 RTE_LOG(INFO, FLOWFWD, "logger started: file %s, level %u\n", path, level);

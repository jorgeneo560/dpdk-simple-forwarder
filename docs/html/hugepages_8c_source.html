32bool hugepages_check(int argc, char **argv) {
33 uint64_t free_bytes = 0;
34 struct dirent *de;
35 DIR *dir;
36
37 for (int i = 1; i < argc && strcmp(argv[i], "--") != 0; i++) {
38 if (strcmp(argv[i], "--no-huge") == 0) {
39 return true;
40 }
41 }
42
43 dir = opendir(HUGEPAGE_SYSFS);
44 if (dir == NULL) {
45 fprintf(stderr, "cannot read %s: %s\n", HUGEPAGE_SYSFS, strerror(errno));
46 return false;
47 }
48
49 while ((de = readdir(dir)) != NULL) {
50 unsigned long size_kb, total, free_pages;
51
52 if (sscanf(de->d_name, "hugepages-%lukB", &size_kb) != 1) {
53 continue;
54 }
55 if (!read_hugepage_count(de->d_name, "nr_hugepages", &total) ||
56 !read_hugepage_count(de->d_name, "free_hugepages", &free_pages)) {
57 continue;
58 }
59 if (total == 0) {
60 continue;
61 }
62
63 fprintf(stderr, "hugepages %lu kB: %lu allocated, %lu free\n", size_kb,
64 total, free_pages);
65 free_bytes += (uint64_t)free_pages * size_kb * 1024;
66 }
67 closedir(dir);
68
69 if (free_bytes == 0) {
70 fprintf(stderr, "no free hugepages. Allocate some, e.g.:\n"
71 " sudo dpdk-hugepages.py -p 2M --setup 2G\n"
72 "or run with --no-huge for testing only.\n");
73 return false;
74 }
75 return true;
76}

79 unsigned int removed = 0;
80 struct dirent *de;

82

90
91 if (strncmp(de->d_name, "rtemap_", 7) != 0) {
92 continue;
93 }
94 snprintf(path, sizeof(path), "%s/%s", mount_dir, de->d_name);
95 fd = open(path, O_RDWR);
96 if (fd < 0) {
97 continue;
98 }
99 // EAL holds a flock on files in use, so a free lock means no process uses
100 // it.
101 if (flock(fd, LOCK_EX | LOCK_NB) == 0 && unlink(path) == 0) {
102 removed++;

104 close(fd);
105 }

77// Returns the number of network devices found, or -1 if the PCI bus cannot be
78// read.
79static int scan_nics(struct nic *nics) {
80 int n = 0;
81 struct dirent *de;
82 DIR *dir = opendir(PCI_SYSFS);
83
84 if (dir == NULL) {
85 fprintf(stderr, "cannot read %s: %s\n", PCI_SYSFS, strerror(errno));
86 return -1;
87 }
88
89 while (n < MAX_NICS && (de = readdir(dir)) != NULL) {
90 char cls[32], vendor[32], device[32];
91 struct nic *nic = &nics[n];
92
93 if (de->d_name[0] == '.') {
94 continue;
95 }
96 if (!read_attr(de->d_name, "class", cls, sizeof(cls)) ||
97 (strtoul(cls, NULL, 16) >> 16) != PCI_CLASS_NET) {
98 continue;
99 }
100 if (!read_attr(de->d_name, "vendor", vendor, sizeof(vendor)) ||
101 !read_attr(de->d_name, "device", device, sizeof(device))) {
102 continue;
103 }
104
105 memset(nic, 0, sizeof(*nic));
106 snprintf(nic->addr, sizeof(nic->addr), "%.15s", de->d_name);
107 nic->vendor = (uint16_t)strtoul(vendor, NULL, 16);
108 nic->device = (uint16_t)strtoul(device, NULL, 16);
109 if (!read_driver(de->d_name, nic->drv, sizeof(nic->drv))) {
110 snprintf(nic->drv, sizeof(nic->drv), "none");
111 }
112 nic->bound = is_dpdk_driver(nic->drv);
113 nic->port = -1;
114 n++;
115 }
116 closedir(dir);
117 return n;
118}

141bool nics_check(int argc, char **argv, unsigned int min_ports,
142 const struct app_config *cfg) {
143 struct nic nics[MAX_NICS];
144 unsigned int ready = 0;
145 int nb_nics;
146
147 for (int i = 1; i < argc && strcmp(argv[i], "--") != 0; i++) {
148 if (strncmp(argv[i], "--vdev", 6) == 0 ||
149 strcmp(argv[i], "--no-pci") == 0) {
150 return true;
151 }
152 }
153
154 nb_nics = scan_nics(nics);
155 if (nb_nics < 0) {
156 fprintf(stderr, "failed to scan network devices\n");
157 return false;
158 }
159
160 // readdir returns the directory entries in no guaranteed order.
161 qsort(nics, (size_t)nb_nics, sizeof(nics[0]), cmp_addr);
162
163 // DPDK probes devices in PCI address order and numbers its ports in that
164 // order.
165 for (int i = 0; i < nb_nics; i++) {
166 if (nics[i].bound) {
167 nics[i].port = (int)ready++;
168 }
169 }
170
171 fprintf(stderr,
172 "PCI network devices (DPDK port ids follow PCI address order):\n");
173 for (int i = 0; i < nb_nics; i++) {
174 print_nic(&nics[i], cfg);
175 }
176
177 if (ready < min_ports) {
178 print_bind_help(ready, min_ports);
179 return false;
180 }
181 return true;
182}

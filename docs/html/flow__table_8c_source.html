43// Written only by the TX thread of the queue, in its own array so that the two
44// threads never write to the same cache line. Never reset: the RX thread notes

46struct flow_tx_entry {

78struct flow_table *flow_table_create(unsigned int id, int socket,
79 const struct app_config *cfg) {
80 char name[RTE_HASH_NAMESIZE];
81 struct rte_hash_parameters params = {
82 .name = name,
83 .entries = cfg->table_size,
84 .key_len = sizeof(struct flow_key),
85 .hash_func = rte_jhash,
86 .socket_id = socket,
87 };
88 struct flow_table *table =
89 rte_zmalloc_socket("flow_table", sizeof(*table), 0, socket);
90
91 if (table == NULL) {
92 RTE_LOG(ERR, FLOWFWD, "flow table %u: out of memory\n", id);
93 return NULL;
94 }
95 table->id = id;
96 table->pass_arp = cfg->pass_arp != 0;
97 table->timeout_cycles = rte_get_tsc_hz() * cfg->flow_timeout;
98 table->cleanup_interval = cfg->cleanup_interval;
99 table->next_cleanup = next_cleanup(table, rte_rdtsc());
100 snprintf(name, sizeof(name), "flows_%u", id);
101 table->hash = rte_hash_create(&params);
102 if (table->hash == NULL) {
103 RTE_LOG(ERR, FLOWFWD, "flow table %u: cannot create %u entries: %s\n", id,
104 cfg->table_size, rte_strerror(rte_errno));
105 rte_free(table);
106 return NULL;
107 }
108 table->entries = rte_zmalloc_socket("flow_entries",
109 cfg->table_size * sizeof(*table->entries),
110 RTE_CACHE_LINE_SIZE, socket);
111 if (table->entries == NULL) {
112 RTE_LOG(ERR, FLOWFWD, "flow table %u: out of memory\n", id);
113 flow_table_free(table);
114 return NULL;
115 }
116 table->tx_entries = rte_zmalloc_socket(
117 "flow_tx_entries", cfg->table_size * sizeof(*table->tx_entries),
118 RTE_CACHE_LINE_SIZE, socket);
119 if (table->tx_entries == NULL) {
120 RTE_LOG(ERR, FLOWFWD, "flow table %u: out of memory\n", id);
121 flow_table_free(table);
122 return NULL;
123 }
124 return table;
125}

137// Adds a received packet to its flow and returns the flow slot, or FLOW_DROP if
138// the table has no room for the flow. *gen is the generation of the slot.
139static int32_t find_or_add_flow(struct flow_table *table,
140 const struct pkt_info *info, uint16_t len,
141 uint64_t now, uint32_t *gen) {
142 struct flow_key key;
143 int slot;
144
145 make_flow_key(&key, info);
146 slot = rte_hash_add_key(table->hash, &key);
147 if (slot < 0) {
148 *gen = 0;
149 return FLOW_DROP;
150 }
151 struct flow_entry *entry;
152 struct flow_tx_entry *tx;
153 entry = &table->entries[slot];
154 tx = &table->tx_entries[slot];
155 // last_seen is 0 only for a slot that is not in use. A new flow gets a new
156 // generation, so a packet of the old flow that is still waiting for the TX
157 // thread is not counted in it.
158 if (entry->last_seen == 0) {
159 atomic_store_explicit(
160 &tx->gen, atomic_load_explicit(&tx->gen, memory_order_relaxed) + 1,
161 memory_order_release);
162 memset(entry, 0, sizeof(*entry));
163 entry->tx_bytes_at_start =
164 atomic_load_explicit(&tx->bytes, memory_order_relaxed);
165 entry->tx_packets_at_start =
166 atomic_load_explicit(&tx->packets, memory_order_relaxed);
167 }
168 entry->last_seen = now;
169 entry->rx_packets++;
170 entry->rx_bytes += len;
171 *gen = (uint32_t)atomic_load_explicit(&tx->gen, memory_order_relaxed);
172 return slot;
173}

175void flow_table_count_sent(struct flow_table *table, int32_t slot, uint32_t gen,
176 uint16_t len) {
177 struct flow_tx_entry *tx = &table->tx_entries[slot];
178
179 // The slot went to another flow while the packet was waiting: not its packet.
180 if ((uint32_t)atomic_load_explicit(&tx->gen, memory_order_acquire) != gen) {
181 return;
182 }
183 // A single writer, so a plain load and store is enough.
184 atomic_store_explicit(
185 &tx->packets,
186 atomic_load_explicit(&tx->packets, memory_order_relaxed) + 1,
187 memory_order_relaxed);
188 atomic_store_explicit(
189 &tx->bytes, atomic_load_explicit(&tx->bytes, memory_order_relaxed) + len,
190 memory_order_relaxed);
191}

193// Removes the flows that were last seen more than timeout cycles before now.
194static void expire_flows(struct flow_table *table, uint64_t now,
195 uint64_t timeout) {
196 const void *key;
197 void *data;
198 uint32_t next = 0;
199 int slot;
200
201 while ((slot = rte_hash_iterate(table->hash, &key, &data, &next)) >= 0) {
202 if (now - table->entries[slot].last_seen <= timeout) {
203 continue;
204 }
205 if (rte_hash_del_key(table->hash, key) >= 0) {
206 table->entries[slot].last_seen = 0;
207 }
208 }
209}

211// TCP and UDP flows are tracked: they have ports, so the 5-tuple is complete.
212// The timeout removes the ones that stop, so no connection end is needed. A
213// packet without ports is a non-first fragment, or another protocol (ICMP).
214static bool has_5tuple(const struct pkt_info *info) {
215 return (info->proto == IPPROTO_TCP || info->proto == IPPROTO_UDP) &&
216 info->src_port != 0 && info->dst_port != 0;
217}

219uint16_t flow_table_filter_burst(const struct flow_table *table,
220 struct rte_mbuf **bufs, uint16_t n,
221 struct burst_meta *meta) {
222 uint16_t kept = 0;
223
224 for (uint16_t i = 0; i < PREFETCH_AHEAD && i < n; i++) {
225 rte_prefetch0(rte_pktmbuf_mtod(bufs[i], void *));
226 }
227
228 for (uint16_t i = 0; i < n; i++) {
229 enum parse_result res;
230 bool keep;
231
232 if (i + PREFETCH_AHEAD < n) {
233 rte_prefetch0(rte_pktmbuf_mtod(bufs[i + PREFETCH_AHEAD], void *));
234 }
235
236 // Parsed straight into the next free slot, which a dropped packet leaves
237 // unused.
238 res = parse_packet(rte_pktmbuf_mtod(bufs[i], const uint8_t *),
239 rte_pktmbuf_data_len(bufs[i]), &meta->info[kept]);
240 keep = res == PARSE_IP || (res == PARSE_ARP && table->pass_arp);
241 if (keep) {
242 meta->len[kept] = (uint16_t)rte_pktmbuf_pkt_len(bufs[i]);
243 bufs[kept++] = bufs[i];
244 } else {
245 rte_pktmbuf_free(bufs[i]);
246 }
247 }
248 meta->count = kept;
249 return kept;
250}

252void flow_table_track(struct flow_table *table, struct burst_meta *meta) {
253 uint64_t now = rte_rdtsc();
254
255 for (uint16_t i = 0; i < meta->count; i++) {
256 if (has_5tuple(&meta->info[i])) {
257 meta->slot[i] = find_or_add_flow(table, &meta->info[i], meta->len[i], now,
258 &meta->gen[i]);
259 } else {
260 meta->slot[i] = FLOW_NO_SLOT;
261 meta->gen[i] = 0;
262 }
263 }
264}

266void flow_table_cleanup(struct flow_table *table, uint64_t now) {
267 if (now < table->next_cleanup) {
268 return;
269 }
270 expire_flows(table, now, table->timeout_cycles);
271 table->next_cleanup = next_cleanup(table, now);
272}

282size_t flow_table_export_next(struct flow_table *table,
283 struct flow_record *recs, size_t max) {
284 const void *key;
285 void *data;
286 size_t n = 0;
287 int slot;
288
289 while (n < max && (slot = rte_hash_iterate(table->hash, &key, &data,
290 &table->export_interval)) >= 0) {
291 const struct flow_key *k = key;
292 const struct flow_entry *e = &table->entries[slot];
293 struct flow_tx_entry *tx = &table->tx_entries[slot];
294 struct flow_record *r = &recs[n++];
295
296 r->family = k->family;
297 r->proto = k->proto;
298 r->src_port = k->src_port;
299 r->dst_port = k->dst_port;
300 r->src_ip = k->src_ip;
301 r->dst_ip = k->dst_ip;
302 r->rx_bytes = e->rx_bytes;
303 r->tx_bytes = atomic_load_explicit(&tx->bytes, memory_order_relaxed) -
304 e->tx_bytes_at_start;
305 r->rx_packets = e->rx_packets;
306 r->tx_packets = atomic_load_explicit(&tx->packets, memory_order_relaxed) -
307 e->tx_packets_at_start;
308 }
309 return n;

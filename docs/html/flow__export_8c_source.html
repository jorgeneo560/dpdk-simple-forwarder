40static void *writer_main(void *arg) {
41 const struct timespec pause = {.tv_sec = 0, .tv_nsec = WRITER_POLL_NS};
42
43 (void)arg;
44 for (;;) {
45 // Read before scanning, so a stop request still gets one last full scan.
46 bool stop = atomic_load(&stopping);
47
48 for (unsigned int i = 0; i < nb_slots; i++) {
49 struct slot *s = &slots[i];
50
51 if (!atomic_load_explicit(&s->ready, memory_order_acquire)) {
52 continue;
53 }
54 for (unsigned int f = 0; f < FILES_PER_QUEUE; f++) {
55 if (!writer_write(&writers[i][f], &s->ts, s->recs, s->n)) {
56 RTE_LOG(ERR, FLOWFWD, "export write failed for queue %u: %s\n", i,
57 strerror(errno));
58 }
59 }
60 atomic_store_explicit(&s->ready, false, memory_order_release);
61 }
62 if (stop) {
63 return NULL;
64 }
65 nanosleep(&pause, NULL);
66 }
67}

98bool flow_export_start(const struct app_config *cfg,
99 const unsigned int *rx_lcores,
100 const unsigned int *tx_lcores, unsigned int n) {
101 if (started) {
102 RTE_LOG(ERR, FLOWFWD, "flow export already started\n");
103 return false;
104 }
105
106 if (n > MAX_QUEUES) {
107 RTE_LOG(ERR, FLOWFWD, "too many queues for flow export: %u\n", n);
108 return false;
109 }
110
111 for (unsigned int i = 0; i < n; i++) {
112 if (!open_file(cfg, rx_lcores[i], &writers[i][FILE_RX])) {
113 close_exporters(i);
114 return false;
115 }
116 if (!open_file(cfg, tx_lcores[i], &writers[i][FILE_TX])) {
117 writer_close(&writers[i][FILE_RX]);
118 close_exporters(i);
119 return false;
120 }
121 atomic_store(&slots[i].ready, false);
122 }
123
124 nb_slots = n;
125 atomic_store(&stopping, false);
126 if (pthread_create(&thread, NULL, writer_main, NULL) != 0) {
127 close_exporters(n);
128 return false;
129 }
130 pthread_setname_np(thread, "flowfwd-export");
131 started = true;
132 return true;
133}

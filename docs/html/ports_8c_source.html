136static bool check_selection(const struct app_config *cfg) {
137 if (rte_eth_dev_count_avail() < NB_PORTS) {
138 RTE_LOG(ERR, FLOWFWD, "need %d ports, found %u\n", NB_PORTS,
139 rte_eth_dev_count_avail());
140 return false;
141 }
142 if (cfg->in_port == cfg->out_port ||
143 !rte_eth_dev_is_valid_port(cfg->in_port) ||
144 !rte_eth_dev_is_valid_port(cfg->out_port)) {
145 RTE_LOG(ERR, FLOWFWD, "invalid port selection: input %u, output %u\n",
146 cfg->in_port, cfg->out_port);
147 return false;
148 }
149 return true;

152// Buffers can be in the RX ring, in the TX ring, in the ring between the RX and
153// the TX thread (one per port, as one per direction) and in a burst being
154// handled. Every lcore also keeps a private cache: an RX and a TX thread per
155// queue, plus the main lcore.
156static bool create_mbuf_pool(const struct app_config *cfg) {
157 const unsigned int nb_mbufs =
158 NB_PORTS * cfg->nb_queues * (RX_DESC + TX_DESC + RING_SIZE + BURST_SIZE) +
159 MBUF_CACHE * (2 * cfg->nb_queues + 1);
160
161 mbuf_pool = rte_pktmbuf_pool_create("MBUF_POOL", nb_mbufs, MBUF_CACHE, 0,
162 MBUF_DATA_LEN, (int)rte_socket_id());
163 if (mbuf_pool == NULL) {
164 RTE_LOG(ERR, FLOWFWD, "mbuf pool creation failed: %s\n",
165 rte_strerror(rte_errno));
166 return false;
167 }
168 RTE_LOG(INFO, FLOWFWD, "mbuf pool: %u mbufs\n", nb_mbufs);
169 return true;
170}

186bool ports_start(const struct app_config *cfg) {
187 if (!check_selection(cfg)) {
188 return false;
189 }
190
191 rss_fill_key(rss_key, sizeof(rss_key), cfg->rss_pattern);
192
193 if (!create_mbuf_pool(cfg)) {
194 return false;
195 }
196
197 if (!start_port(cfg->in_port, cfg)) {
198 RTE_LOG(ERR, FLOWFWD, "cannot initialise input port %u\n", cfg->in_port);
199 return false;
200 }
201 if (!start_port(cfg->out_port, cfg)) {
202 RTE_LOG(ERR, FLOWFWD, "cannot initialise output port %u\n", cfg->out_port);
203 return false;
204 }
205
206 return log_port_roles(cfg);
207}

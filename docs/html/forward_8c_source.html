81// Sets the destination MAC of the IP packets of a burst. ARP frames (no IP
82// family) are left alone.
83static void rewrite_dst_mac(struct rte_mbuf **bufs,
84 const struct burst_meta *meta,
85 const struct rte_ether_addr *mac) {
86 for (uint16_t i = 0; i < meta->count; i++) {
87 if (meta->info[i].family != 0) {
88 struct rte_ether_hdr *eth =
89 rte_pktmbuf_mtod(bufs[i], struct rte_ether_hdr *);
90
91 rte_ether_addr_copy(mac, &eth->dst_addr);
92 }
93 }
94}

96// Counters are updated once per burst, not per packet. A counter has a single
97// writer, so a plain load and store is enough, without a locked instruction.
98static inline void add_stat(atomic_uint_fast64_t *counter, uint64_t n) {
99 atomic_store_explicit(counter,
100 atomic_load_explicit(counter, memory_order_relaxed) + n,
101 memory_order_relaxed);
102}

104// Makes a ring item of each packet, and frees the packets whose flow did not
105// fit in the table. Returns how many items were made.
106static unsigned int make_items(struct rte_mbuf **bufs,
107 const struct burst_meta *meta,
108 struct tx_item *items) {
109 unsigned int n = 0;
110
111 for (uint16_t i = 0; i < meta->count; i++) {
112 if (meta->slot[i] == FLOW_DROP) {
113 rte_pktmbuf_free(bufs[i]);
114 } else {
115 items[n++] = (struct tx_item){.mbuf = bufs[i],
116 .slot = meta->slot[i],
117 .gen = meta->gen[i],
118 .len = meta->len[i]};
119 }
120 }
121 return n;
122}

124// Puts the items in the ring; the ones that do not fit are freed and counted.
125static void queue_items(struct rx_stats *stats, struct rte_ring *ring,
126 struct tx_item *items, unsigned int n) {
127 unsigned int queued =
128 rte_ring_enqueue_burst_elem(ring, items, sizeof(items[0]), n, NULL);
129
130 for (unsigned int i = queued; i < n; i++) {
131 rte_pktmbuf_free(items[i].mbuf);
132 }
133 add_stat(&stats->ring_drops, n - queued);
134}

136// RX thread: reads a burst from one interface, filters and counts it, and
137// queues the packets for the TX thread. Returns how many packets were read.
138static uint16_t receive_burst(struct queue_args *args, uint16_t from,
139 struct rte_ring *ring,
140 const struct rte_ether_addr *dst_mac) {
141 struct rx_stats *stats = &args->rx_stats;
142 struct rte_mbuf *bufs[BURST_SIZE];
143 struct burst_meta meta;
144 struct tx_item items[BURST_SIZE];
145 uint16_t nb_rx = rte_eth_rx_burst(from, args->queue, bufs, BURST_SIZE);
146 uint16_t nb_kept;
147 unsigned int nb_items;
148
149 if (nb_rx == 0) {
150 return 0;
151 }
152 add_stat(&stats->rx_packets, nb_rx);
153
154 nb_kept = flow_table_filter_burst(args->table, bufs, nb_rx, &meta);
155 if (nb_kept == 0) {
156 return nb_rx;
157 }
158 if (dst_mac != NULL) {
159 rewrite_dst_mac(bufs, &meta, dst_mac);
160 }
161 flow_table_track(args->table, &meta);
162
163 nb_items = make_items(bufs, &meta, items);
164 add_stat(&stats->table_drops, nb_kept - nb_items);
165 queue_items(stats, ring, items, nb_items);
166 return nb_rx;
167}

169// TX thread: takes a burst from a ring, sends it and counts the sent packets in
170// their flows. Returns how many packets were taken.
171static uint16_t send_burst(struct queue_args *args, struct rte_ring *ring,
172 uint16_t to) {
173 struct tx_stats *stats = &args->tx_stats;
174 struct tx_item items[BURST_SIZE];
175 struct rte_mbuf *bufs[BURST_SIZE];
176 unsigned int n = rte_ring_dequeue_burst_elem(ring, items, sizeof(items[0]),
177 BURST_SIZE, NULL);
178 uint16_t nb_tx;
179
180 if (n == 0) {
181 return 0;
182 }
183 for (unsigned int i = 0; i < n; i++) {
184 bufs[i] = items[i].mbuf;
185 }
186 nb_tx = rte_eth_tx_burst(to, args->queue, bufs, (uint16_t)n);
187 add_stat(&stats->tx_packets, nb_tx);
188
189 // Only the saved slot and length are used: a sent packet is no longer ours.
190 for (uint16_t i = 0; i < nb_tx; i++) {
191 if (items[i].slot != FLOW_NO_SLOT) {
192 flow_table_count_sent(args->table, items[i].slot, items[i].gen,
193 items[i].len);
194 }
195 }
196 if (nb_tx < n) {
197 rte_pktmbuf_free_bulk(&bufs[nb_tx], n - nb_tx);
198 add_stat(&stats->tx_drops, n - nb_tx);
199 }
200 return (uint16_t)n;
201}

218static void export_pass_step(struct export_pass *pass, uint16_t queue,
219 struct flow_table *table) {
220 struct flow_record *recs = flow_export_begin(queue);
221 size_t n;
222
223 if (recs == NULL) {
224 return; // the writer is still busy with the previous chunk
225 }
226 n = flow_table_export_next(table, recs, FLOW_EXPORT_MAX);
227 if (n == 0) {
228 pass->active = false;
229 return;
230 }
231 flow_export_commit(queue, &pass->ts, n);
232}

234// Adds the time since the previous call to the load counters of a thread, as
235// busy time if it handled packets. Returns the current time.
236static uint64_t count_cycles(uint64_t *last, bool handled,
237 atomic_uint_fast64_t *busy_cycles,
238 atomic_uint_fast64_t *total_cycles) {
239 uint64_t now = rte_rdtsc();
240
241 add_stat(total_cycles, now - *last);
242 if (handled) {
243 add_stat(busy_cycles, now - *last);
244 }
245 *last = now;
246 return now;
247}

249static int rx_main(void *arg) {
250 struct queue_args *args = arg;
251 struct rx_stats *stats = &args->rx_stats;
252 uint64_t last, next_export;
253 struct export_pass export_pass = {.active = false};
254
255 RTE_LOG(DEBUG, FLOWFWD, "RX thread started: queue %u, ports %u <-> %u\n",
256 args->queue, args->in_port, args->out_port);
257 last = rte_rdtsc();
258 next_export = last + export_cycles;
259 while (!atomic_load_explicit(&rx_quit, memory_order_relaxed)) {
260 unsigned int handled = receive_burst(
261 args, args->in_port, args->in_to_out_ring, args->forward_mac);
262 uint64_t now;
263
264 handled += receive_burst(args, args->out_port, args->out_to_in_ring,
265 args->reverse_mac);
266
267 now = count_cycles(&last, handled > 0, &stats->busy_cycles,
268 &stats->total_cycles);
269 if (!export_pass.active && now >= next_export) {
270 export_pass_start(&export_pass, args->table);
271 next_export = now + export_cycles;
272 }
273 if (export_pass.active) {
274 export_pass_step(&export_pass, args->queue, args->table);
275 }
276 flow_table_cleanup(args->table, now);
277 atomic_store_explicit(&stats->flows, flow_table_count(args->table),
278 memory_order_relaxed);
279 }
280 RTE_LOG(DEBUG, FLOWFWD, "RX thread stopped: queue %u\n", args->queue);
281 return 0;
282}

292static int tx_main(void *arg) {
293 struct queue_args *args = arg;
294 struct tx_stats *stats = &args->tx_stats;
295 uint64_t last;
296
297 RTE_LOG(DEBUG, FLOWFWD, "TX thread started: queue %u, ports %u <-> %u\n",
298 args->queue, args->in_port, args->out_port);
299 last = rte_rdtsc();
300 while (!atomic_load_explicit(&tx_quit, memory_order_relaxed) ||
301 !rings_empty(args)) {
302 unsigned int handled =
303 send_burst(args, args->in_to_out_ring, args->out_port);
304
305 handled += send_burst(args, args->out_to_in_ring, args->in_port);
306 count_cycles(&last, handled > 0, &stats->busy_cycles, &stats->total_cycles);
307 }
308 RTE_LOG(DEBUG, FLOWFWD, "TX thread stopped: queue %u\n", args->queue);
309 return 0;
310}

423// The worker lcores are taken in pairs: the first of a pair is the RX thread of
424// a queue and the second one its TX thread.
425static bool pick_lcores(unsigned int nb) {
426 const unsigned int needed = 2 * nb;
427 unsigned int lcore, found = 0;
428
429 if (rte_lcore_count() - 1 < needed) {
430 RTE_LOG(ERR, FLOWFWD,
431 "need %u worker lcores (an RX and a TX thread per queue) plus the "
432 "main lcore, have %u in total\n",
433 needed, rte_lcore_count());
434 return false;
435 }
436 RTE_LCORE_FOREACH_WORKER(lcore) {
437 if (found == needed) {
438 break;
439 }
440 if (found % 2 == 0) {
441 queue_args[found / 2].rx_lcore = lcore;
442 } else {
443 queue_args[found / 2].tx_lcore = lcore;
444 }
445 found++;
446 }
447 return true;
448}

469 }

473static bool launch_threads(const struct app_config *cfg,
474 const struct rte_ether_addr *forward_mac,
475 const struct rte_ether_addr *reverse_mac) {
476 for (uint16_t queue = 0; queue < nb_queues; queue++) {
477 struct queue_args *args = &queue_args[queue];
478
479 args->queue = queue;
480 args->in_port = cfg->in_port;
481 args->out_port = cfg->out_port;
482 args->forward_mac = forward_mac;
483 args->reverse_mac = reverse_mac;
484 // The consumer is started first, so the ring never fills before it runs.
485 if (rte_eal_remote_launch(tx_main, args, args->tx_lcore) != 0 ||
486 rte_eal_remote_launch(rx_main, args, args->rx_lcore) != 0) {
487 RTE_LOG(ERR, FLOWFWD, "cannot launch the threads of queue %u\n", queue);
488 forward_stop();
489 return false;
490 }
491 RTE_LOG(INFO, FLOWFWD,
492 "queue %u: RX thread on lcore %u, TX thread on lcore %u\n", queue,
493 args->rx_lcore, args->tx_lcore);
494 }
495 return true;
496}

498bool forward_start(const struct app_config *cfg) {
499 const struct rte_ether_addr *forward_mac =
500 rewrite_target(cfg->forward_dst_mac, &forward_mac_storage);
501 const struct rte_ether_addr *reverse_mac =
502 rewrite_target(cfg->reverse_dst_mac, &reverse_mac_storage);
503
504 if (!pick_lcores(cfg->nb_queues)) {
505 return false;
506 }
507 nb_queues = cfg->nb_queues;
508 atomic_store(&rx_quit, false);
509 atomic_store(&tx_quit, false);
510 export_cycles = rte_get_tsc_hz() * cfg->export_interval;
511
512 if (!create_resources(cfg)) {
513 return false;
514 }
515 log_rewrite(forward_mac, reverse_mac);
516 return launch_threads(cfg, forward_mac, reverse_mac);
517}

540// The last export, made when all threads have stopped, so the flows of the
541// final interval are in the files with their final counters.
542static void final_export(void) {
543 for (unsigned int q = 0; q < nb_queues; q++) {
544 struct export_pass pass;
545
546 export_pass_start(&pass, queue_args[q].table);
547 while (pass.active) {
548 export_pass_step(&pass, (uint16_t)q, queue_args[q].table);

554// Order: the RX threads stop reading, the TX threads send what is left in the
555// rings and stop, then the last export is written.
556void forward_stop(void) {
557 atomic_store(&rx_quit, true);
558 for (unsigned int q = 0; q < nb_queues; q++) {
559 rte_eal_wait_lcore(queue_args[q].rx_lcore);
560 }
561 atomic_store(&tx_quit, true);
562 for (unsigned int q = 0; q < nb_queues; q++) {
563 rte_eal_wait_lcore(queue_args[q].tx_lcore);
564 }
565 log_queue_summary();
566 final_export();
567 flow_export_stop();
568 free_rings();
569 free_flow_tables();
570}

614// Share of the time since the previous call that a thread spent handling
615// packets, in percent. The previous values are kept in prev_busy/prev_total.
616static unsigned int load_percent(atomic_uint_fast64_t *busy_cycles,
617 atomic_uint_fast64_t *total_cycles,
618 uint64_t *prev_busy, uint64_t *prev_total) {
619 uint64_t busy = atomic_load_explicit(busy_cycles, memory_order_relaxed);
620 uint64_t total = atomic_load_explicit(total_cycles, memory_order_relaxed);
621 uint64_t d_total = total - *prev_total;
622 unsigned int pct =
